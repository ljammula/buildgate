sys.path.insert(0, ".")
ran = failed = 0
for path in sorted(glob.glob(".oracle/test_oracle_*.py")):
    spec = importlib.util.spec_from_file_location("oracle_mod", path)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    for name in sorted(vars(mod)):
        fn = getattr(mod, name)
        if name.startswith("test") and callable(fn):
            ran += 1
            try:
                fn()
            except Exception:
                failed += 1
                traceback.print_exc()
print("oracle tests: %d ran, %d failed" % (ran, failed))
sys.exit(1 if failed or not ran else 0)
